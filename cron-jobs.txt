# BlackHornet Cron Jobs Configuration
# Format: schedule|action|command|description

# Watchdog - Run every 5 minutes
*/5 * * * *|run|cd /sdcard/dhaherlabs/blackhornet-worktree && python3 scripts/bh_watchdog.py|BH Doctor health monitor & self-heal

# Auto-Export Sessions - Run every 10 minutes  
*/10 * * * *|run|cd /sdcard/dhaherlabs/blackhornet-worktree && python3 -c "from scripts.bh_watchdog import SessionExporter; e = SessionExporter(); print(f'Exported {len(e.check_new_sessions())} sessions')"|Auto-export sessions

# Auto-Push to Repos - Run every 30 minutes (only if changes)
*/30 * * * *|run|cd /sdcard/dhaherlabs/blackhornet-worktree && bash scripts/auto-push.sh "[AUTO] Scheduled push - $(date '+%Y-%m-%d %H:%M')"|Auto-push to Codeberg + GitLab

# Daily full backup - Run at 2 AM every day
0 2 * * *|run|cd /sdcard/dhaherlabs/blackhornet-worktree && tar -czf /sdcard/dhaherlabs/backups/blackhornet-$(date '+%Y%m%d').tar.gz .|Daily full backup

# Weekly audit - Run every Sunday at 3 AM
0 3 * * 0|run|cd /sdcard/dhaherlabs/blackhornet-worktree && python3 -c "print('Weekly audit placeholder')"|Weekly comprehensive audit

# Graphify cleanup - Run daily at 4 AM
0 4 * * *|run|find /sdcard/dhaherlabs/blackhornet-worktree/.bh/graph -name '*.json' -mtime +30 -delete|Clean old graph files (30+ days)

# Log rotation - Run daily at 5 AM
0 5 * * *|run|find /sdcard/dhaherlabs/blackhornet-worktree/logs -name '*.log.*' -mtime +7 -delete|Clean old logs (7+ days)
